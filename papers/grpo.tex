\section*{Section 2: Preliminaries}

\begin{itemize}
    \item The policy LLM: $\pi_{\theta}$
    \item A group of G complete responses: $\{x_{1},...,x_{G}\}$
    \item A single scalar reward for response $x_i$: $r_{i}$
    \item The response-level advantage, calculated using a z-score normalization of rewards:
        $$\hat{A}_{i}=\frac{r_{i}-\text{mean}(r_{1},...,r_{G})}{\text{std}(r_{1},...,r_{G})+\epsilon_{\text{norm}}}$$
    \item A small constant for numerical stability: $\epsilon_{\text{norm}}$
    \item The clipped surrogate objective for the policy update in Group Relative Policy Optimization (GRPO):
        $$L_{\text{GRPO}}(\theta) = -\frac{1}{G}\sum_{i=1}^{G} \min\left(\frac{\pi_{\theta}(x_i)}{\pi_{\theta_{\text{old}}}(x_i)}\hat{A}_i, \text{clip}\left(\frac{\pi_{\theta}(x_i)}{\pi_{\theta_{\text{old}}}(x_i)}, 1-\epsilon, 1+\epsilon\right)\hat{A}_i\right) + \beta D_{\text{KL}}(\pi_{\theta}||\pi_{\theta_{\text{old}}})$$
    \item A rule-based verifier function: $v:\mathcal{X}\rightarrow\{0,1\}$
    \item The binary reward for a generation $x_i$:
        $$r_i = v(x_i) = \begin{cases} 1, & \text{if } x_i \text{ satisfies a task-specific correctness check,} \\ 0, & \text{otherwise.} \end{cases}$$
\end{itemize}

\section*{Section 3: Method}

\begin{itemize}
    \item Challenger model: $Q_{\theta}$
    \item Solver model: $S_{\phi}$
    \item Final scalar reward used in GRPO advantage calculation: $r_{ij}$
    \item A generated question: $x$
    \item Number of responses from the Solver: $m$
    \item A set of $m$ responses for a question $x$: $\{y_{1},...,y_{m}\}$
    \item The pseudo-label (most frequent response): $\overline{y}(x)$
    \item The Solver's empirical accuracy for a question $x$:
        $$\hat{p}(x;S_{\phi})=\frac{1}{m}\sum_{j=1}^{m}\mathbb{I}\{y_{j}=\tilde{y}(x)\}$$
    \item The uncertainty reward function:
        $$r_{\text{uncertainty}}(x;\phi)=1-2|\hat{p}(x;S_{\phi})-\frac{1}{2}|$$
    \item Pairwise distance between questions using BLEU score: $d_{ij}=1-\text{BLEU}(x_{i},x_{j})$
    \item Condition for grouping questions into clusters: $d_{ij}<\tau_{\text{BLEU}}$
    \item A set of question clusters: $\mathcal{C}=\{C_{1},...,C_{K}\}$
    \item The repetition penalty for a question $x_i$ in cluster $C_k$:
        $$r_{\text{rep}}(x_i) = \lambda \frac{|C_k|}{B}$$
    \item Scaling factor for repetition penalty (set to 1 in experiments): $\lambda$
    \item Batch size: $B$
    \item The composite reward for a valid question $x_i$:
        $$r_{i}=\max(0,r_{\text{uncertainty}}(x_{i};\phi)-r_{\text{rep}}(x_{i}))$$
    \item The GRPO loss for the Challenger: $\mathcal{L}_{\text{GRPO}}(\theta)$
    \item Sampling a candidate question from the Challenger's policy: $x_{i}\sim Q_{\theta}(\cdot|p_{0})$
    \item A question-answer pair: $(x_{i},\overline{y}_{i})$
    \item The training set for the Solver: $\mathcal{S}$
    \item Filtering condition for adding a question to the training set: $|\hat{p}_{i}-\frac{1}{2}|\le\delta$
    \item Binary reward for a Solver-generated answer $x_j$ for a question with pseudo-label $\overline{y}_{i}$:
        $$r_j = \begin{cases} 1, & \text{if } x_j \text{ is identical to the pseudo-label } \overline{y}_{i}, \\ 0, & \text{otherwise.} \end{cases}$$
    \item The GRPO loss for the Solver: $\mathcal{L}_{\text{GRPO}}(\phi)$
    \item The Solver's success probability: $p$
    \item KL divergence between the current Solver policy ($S_{\phi}$) and an optimal policy ($S^{*}$): $\mathbb{D}_{\text{KL}}(S_{\phi}||S^{*})$
    \item Lower bound on the KL divergence, where $\beta$ is a temperature parameter:
        $$D_{\text{KL}}(S_{\phi}||S^{*}) \geq \frac{p(1-p)}{2\beta^2}$$
    \item Condition that maximizes the reward variance (and thus the lower bound): $\hat{p}=0.5$
\end{itemize}

\section*{Section 4: Experiments}

\begin{itemize}
    \item Number of candidate questions generated by the Challenger: $N=8,000$
    \item Number of answers sampled from the Solver for dataset construction and uncertainty reward calculation: $m=10$
    \item Filtering parameter for dataset construction: $\delta=0.25$
    \item Clustering distance threshold for the repetition penalty: $\tau_{\text{BLEU}}=0.5$
\end{itemize}

\section*{Appendix A: Experiment Details}

\begin{itemize}
    \item Learning Rate: $1\times10^{-6}$
    \item Weight Decay: $1\times10^{-2}$
    \item KL Penalty Coefficient: $1\times10^{-2}$
    \item Pairwise distance calculation for repetition penalty: $d_{ij}=1-\text{BLEU}(x_{i},x_{j})$
    \item Final repetition penalty calculation for a question $x_i$ in cluster $C_k$:
        $$P_{\text{rep}}(x_i) = \frac{|C_k|}{B}$$
    \item The number of questions in cluster $C_k$: $|C_{k}|$
    \item The total number of questions in the batch: $B$
\end{itemize}
